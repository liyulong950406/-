# 聚宽因子库全面介绍

## 一、聚宽因子库概述

### 1.1 什么是聚宽因子

聚宽因子是预定义的量化指标，用于衡量股票的各种特征，涵盖估值、质量、成长、技术、风险等各个维度。

### 1.2 因子分类

聚宽因子库包含300多个因子，主要分为以下几类：

```

def get_growth_factors(stocks, start_date, end_date):
    """
    计算成长因子（需要比较期数据）
    """
    # 获取两个时间点的财务数据
    q = query(
        income.code,
        income.total_operating_revenue,  # 营业收入
        income.net_profit,               # 净利润
        income.operating_profit          # 营业利润
    ).filter(
        income.code.in_(stocks)
    )
    
    # 获取当前期数据
    current_data = get_fundamentals(q, statDate=end_date[:4])
    
    # 获取上年同期数据
    last_year = str(int(end_date[:4]) - 1)
    last_data = get_fundamentals(q, statDate=last_year)
    
    # 计算增长率
    growth_rates = {}
    for stock in stocks:
        if stock in current_data['code'].values and stock in last_data['code'].values:
            curr_revenue = current_data[current_data['code'] == stock]['total_operating_revenue'].values[0]
            last_revenue = last_data[last_data['code'] == stock]['total_operating_revenue'].values[0]
            
            if last_revenue != 0:
                revenue_growth = (curr_revenue - last_revenue) / last_revenue
                growth_rates[stock] = revenue_growth
    
    return growth_rates
```

```
def calculate_technical_factors(stocks, end_date, period=20):

    计算技术因子
    """
    import talib
    
    tech_factors = {}
    
    for stock in stocks[:10]:  # 示例：只计算前10只
        # 获取价格数据
        prices = get_price(stock, 
                          end_date=end_date, 
                          count=period+1,  # 多取一天用于计算
                          fields=['close', 'volume', 'high', 'low'])
        
        close = prices['close'].values
        high = prices['high'].values
        low = prices['low'].values
        volume = prices['volume'].values
        
        if len(close) >= period:
            # 计算RSI
            rsi = talib.RSI(close, timeperiod=14)[-1]
            
            # 计算波动率
            returns = np.diff(np.log(close))
            volatility = np.std(returns) * np.sqrt(252)  # 年化波动率

            turnover = volume[-1] * close[-1]  # 近似成交额
            
            tech_factors[stock] = {
                'rsi': rsi,
                'volatility': volatility,
                'turnover': turnover
            }
    
    return pd.DataFrame(tech_factors).T
```

```
def factor_based_stock_selection(date, factor_name, top_n=10, universe='沪深300'):
    """
    基于单因子的选股策略
    """
    # 获取股票池
    if universe == '沪深300':
        stocks = get_index_stocks('000300.XSHG', date=date)
    else:
        stocks = get_all_securities(['stock'], date=date).index.tolist()
    
    # 获取因子值
    factor_values = get_factor_values(
        securities=stocks,
        factors=[factor_name],
        count=1,
        end_date=date
    )[factor_name]
    
    if factor_values is None:
        return []
    
    # 根据因子逻辑排序
    # 估值因子：通常选择低估值（PE, PB等越小越好）
    if factor_name in ['pe_ratio', 'pb_ratio', 'ps_ratio']:
        # 去除异常值
        factor_values = factor_values[factor_values > 0]
        selected = factor_values.nsmallest(top_n).index.tolist()
    # 质量因子：通常选择高质量（ROE, ROA等越大越好）
    elif factor_name in ['roe', 'roa', 'gross_profit_margin']:
        selected = factor_values.nlargest(top_n).index.tolist()
    # 动量因子：选择正动量
    elif factor_name in ['returns', 'momentum']:
        selected = factor_values.nlargest(top_n).index.tolist()
    else:
        # 默认选择因子值最大的
        selected = factor_values.nlargest(top_n).index.tolist()
    
    return selected
```

### 3.2 风险控制应用

```
def risk_control_with_factors(portfolio, date):
    """
    使用因子进行风险控制
    """
    risk_factors = ['volatility', 'beta', 'max_drawdown']
    
    portfolio_risk = {}
    
    for factor in risk_factors:
        # 获取组合中各股票的因子值
        factor_values = get_factor_values(
            securities=portfolio,
            factors=[factor],
            count=1,
            end_date=date
        )[factor]
        
        if factor_values is not None:
            # 计算组合平均风险
            portfolio_risk[factor] = factor_values.mean()
            
            # 风险警示
            if factor == 'volatility' and portfolio_risk[factor] > 0.3:
                print(f"警告：组合波动率过高: {portfolio_risk[factor]:.2%}")
            elif factor == 'beta' and abs(portfolio_risk[factor] - 1) > 0.5:
                print(f"警告：组合Beta偏离较大: {portfolio_risk[factor]:.2f}")
    
    return portfolio_risk
```

```
def performance_attribution(returns, benchmark_returns, factors_list, date):
    """
    使用因子进行绩效归因
    """
    from sklearn.linear_model import LinearRegression
    
    # 获取因子暴露
    factor_exposures = {}
    
    for factor in factors_list:
        # 获取全市场因子值
        all_stocks = get_all_securities(['stock'], date=date).index.tolist()
        factor_data = get_factor_values(
            securities=all_stocks,
            factors=[factor],
            count=1,
            end_date=date
        )[factor]
        
        if factor_data is not None:
            # 计算组合的因子暴露（组合内股票的平均因子值）
            # 这里简化处理，实际需要更复杂的计算
            factor_exposures[factor] = factor_data.mean()
    
    # 构建因子模型
    X = pd.DataFrame(factor_exposures, index=[0])
    y = returns - benchmark_returns  # 超额收益
    
    if not y.empty:
        # 简单线性回归
        model = LinearRegression()
        model.fit(X, y)
        
        attribution = pd.DataFrame({
            'factor': factors_list,
            'exposure': X.values[0],
            'contribution': model.coef_
        })
        
        return attribution
```

### 4.1 多因子选股策略

```
class MultiFactorStrategy:
    """
    多因子选股策略示例
    """
    def __init__(self):
        # 定义使用的因子及其方向
        self.factors_config = {
            'valuation': {
                'pe_ratio': 'asc',      # 低PE
                'pb_ratio': 'asc',      # 低PB
                'dividend_yield': 'desc'  # 高股息
            },
            'quality': {
                'roe': 'desc',          # 高ROE
                'gross_profit_margin': 'desc'  # 高毛利
            },
            'growth': {
                'inc_revenue': 'desc'   # 高增长
            }
        }
    
    def calculate_composite_score(self, stocks, date):
        """
        计算综合因子得分
        """
        all_scores = {}
        
        for category, factors in self.factors_config.items():
            for factor_name, direction in factors.items():
                # 获取因子值
                factor_values = get_factor_values(
                    securities=stocks,
                    factors=[factor_name],
                    count=1,
                    end_date=date
                )[factor_name]
                
                if factor_values is not None:
                    # 处理缺失值
                    factor_values = factor_values.fillna(factor_values.median())
                    
                    # 去极值
                    factor_values = self.winsorize(factor_values)
                    
                    # 标准化
                    factor_values = (factor_values - factor_values.mean()) / factor_values.std()
                    
                    # 根据方向调整
                    if direction == 'asc':
                        factor_values = -factor_values  # 越小越好，取负值
                    
                    all_scores[factor_name] = factor_values
        
        # 等权合成
        score_df = pd.DataFrame(all_scores)
        composite_score = score_df.mean(axis=1)
        
        return composite_score.sort_values(ascending=False)
    
    def winsorize(self, series, n=3):
        """
        去极值处理
        """
        mean = series.mean()
        std = series.std()
        lower = mean - n * std
        upper = mean + n * std
        return series.clip(lower, upper)
    
    def select_stocks(self, date, top_n=20, universe='沪深300'):
        """
        选股
        """
        # 获取股票池
        if universe == '沪深300':
            stocks = get_index_stocks('000300.XSHG', date=date)
        else:
            stocks = get_all_securities(['stock'], date=date).index.tolist()[:500]  # 限制数量
        
        # 计算综合得分
        composite_score = self.calculate_composite_score(stocks, date)
        
        # 选择得分最高的股票
        selected = composite_score.head(top_n).index.tolist()
        
        return selected, composite_score
```

### 4.2 因子轮动策略

```
class FactorRotationStrategy:
    """
    因子轮动策略
    """
    def __init__(self):
        self.factors_pool = ['pe_ratio', 'pb_ratio', 'roe', 'inc_revenue', 'returns']
        self.lookback_period = 60  # 回看期
        self.holding_period = 20   # 持有期
    
    def calculate_factor_performance(self, date):
        """
        计算各因子近期表现
        """
        end_date = date
        start_date = get_trade_days(end_date=end_date, count=self.lookback_period)[0]
        
        factor_performance = {}
        
        for factor in self.factors_pool:
            # 获取因子值
            all_stocks = get_index_stocks('000300.XSHG', date=start_date)
            factor_values = get_factor_values(
                securities=all_stocks,
                factors=[factor],
                count=1,
                end_date=start_date
            )[factor]
            
            if factor_values is not None:
                # 按因子分组
                groups = pd.qcut(factor_values, 5, labels=False, duplicates='drop')
                
                # 获取未来收益
                future_returns = {}
                for i in range(5):
                    group_stocks = groups[groups == i].index.tolist()
                    if group_stocks:
                        prices = get_price(group_stocks,
                                         start_date=start_date,
                                         end_date=end_date,
                                         fields=['close'])
                        
                        if not prices.empty:
                            # 计算组合收益
                            start_prices = prices['close'].iloc[0]
                            end_prices = prices['close'].iloc[-1]
                            returns = (end_prices / start_prices - 1).mean()
                            future_returns[f'group_{i}'] = returns
                
                # 计算多空收益
                if 'group_4' in future_returns and 'group_0' in future_returns:
                    long_short = future_returns['group_4'] - future_returns['group_0']
                    factor_performance[factor] = long_short
        
        return factor_performance
    
    def select_best_factor(self, date):
        """
        选择表现最好的因子
        """
        performance = self.calculate_factor_performance(date)
        
        if performance:
            best_factor = max(performance.items(), key=lambda x: x[1])[0]
            return best_factor
        else:
            return 'pe_ratio'  # 默认使用PE
```

### 5.1 因子预处理

```
def preprocess_factors(factor_data):
    """
    因子预处理
    """
    processed = factor_data.copy()
    
    # 1. 处理缺失值
    processed = processed.fillna(processed.median())
    
    # 2. 去极值
    def winsorize_series(series, n=3):
        mean = series.mean()
        std = series.std()
        lower = mean - n * std
        upper = mean + n * std
        return series.clip(lower, upper)
    
    processed = processed.apply(winsorize_series)
    
    # 3. 标准化
    processed = (processed - processed.mean()) / processed.std()
    
    # 4. 中性化处理（可选）
    # 可对行业、市值等因子进行中性化
    
    return processed
```

### 5.2 因子有效性检验

```
def validate_factor(factor_name, start_date, end_date, universe='沪深300'):
    """
    检验因子有效性
    """
    dates = get_trade_days(start_date=start_date, end_date=end_date)
    ic_series = []
    
    for i, date in enumerate(dates[:-20]):  # 预留20天计算未来收益
        # 获取股票池
        stocks = get_index_stocks('000300.XSHG', date=date) if universe == '沪深300' \
                else get_all_securities(['stock'], date=date).index.tolist()[:500]
        
        # 获取因子值
        factor_values = get_factor_values(
            securities=stocks,
            factors=[factor_name],
            count=1,
            end_date=date
        )[factor_name]
        
        if factor_values is not None:
            # 获取未来20天收益
            future_date = dates[i+20] if i+20 < len(dates) else dates[-1]
            
            returns = {}
            for stock in factor_values.dropna().index:
                price = get_price(stock, 
                                start_date=date, 
                                end_date=future_date,
                                fields=['close'])
                if not price.empty and len(price) > 1:
                    ret = price['close'].iloc[-1] / price['close'].iloc[0] - 1
                    returns[stock] = ret
            
            # 计算IC
            common_stocks = set(factor_values.dropna().index) & set(returns.keys())
            if len(common_stocks) > 10:
                factor_series = factor_values[list(common_stocks)]
                return_series = pd.Series(returns)[list(common_stocks)]
                
                ic = factor_series.rank().corr(return_series.rank())
                ic_series.append(ic)
    
    # 分析IC
    if ic_series:
        ic_mean = np.mean(ic_series)
        ic_std = np.std(ic_series)
        ic_ir = ic_mean / ic_std if ic_std > 0 else 0
        positive_ratio = sum(1 for x in ic_series if x > 0) / len(ic_series)
        
        return {
            'ic_mean': ic_mean,
            'ic_std': ic_std,
            'ic_ir': ic_ir,
            'positive_ratio': positive_ratio,
            'ic_series': ic_series
        }
    
    return None
```

## 六、最佳实践建议

   **因子选择**：

     选择逻辑清晰的因子

   **回测验证**：

     考虑交易成本
